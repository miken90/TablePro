public.sp_rebuild_indexes(table_name text, fill_factor integer)

-- procedure definition, read-only CREATE OR REPLACE PROCEDURE public.sp_rebuild_indexes( table_name text DEFAULT NULL, fill_factor integer DEFAULT 90 ) LANGUAGE plpgsql AS $$ DECLARE idx record; BEGIN FOR idx IN SELECT indexname, tablename FROM pg_indexes WHERE schemaname = 'public' AND (table_name IS NULL OR tablename = table_name) LOOP EXECUTE format('REINDEX INDEX %I', idx.indexname); END LOOP; END; $$;
